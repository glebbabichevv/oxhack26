# Horizon Model v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A candidate model for Horizon that picks the most accurate of three calibration models on validation, trades with real Polymarket fees and a one-bankroll exposure cap, and reports one honest out-of-time test.

**Architecture:** `features.py` turns `data/` into one feature table (no model code). `models.py` holds M0/M1/M2 behind one `fit / predict → (q, lo, hi)` interface. `decision.py` holds cost, decide, earliest-horizon policy and the exposure cap. `train.py` runs the playoff on validation, tunes the betting knobs on validation, scores the test once and writes `results/*.json` in the same shapes as `backend/results/`, reusing `backend/model.py`'s metric helpers read-only.

**Tech Stack:** Python 3.11, pandas, numpy, scikit-learn 1.9 (LogisticRegression, StandardScaler, HistGradientBoostingClassifier), pytest 9.

**Spec:** `docs/superpowers/specs/2026-10-24-model-v2-design.md`

## Global Constraints

- Work only in `integrations/` (Role C folder). `backend/model.py` is imported read-only, never edited.
- Out-of-time cutoff `2026-05-18`; test = events whose last market resolved on/after the cutoff, and only their markets that resolved on/after it (same rule as `backend/model.py:split_by_time`).
- Validation = last 20% of train **events** by event end time; never split an event.
- Tradable band 3–97¢ (`0.03 ≤ p ≤ 0.97`); outside it q = p and NO_TRADE.
- `cost = max(spread/2, 0.01) + fees_on × fee_rate[domain] × p × (1 − p)`; fee rates: sports 0.05, crypto 0.07, politics 0.04, science_tech 0.04, economics 0.05, culture 0.05, other 0.05, geopolitics 0. Entry only.
- Stake caps: 5% per trade, 10% per event, total open stakes ≤ 1.0 bankroll. P&L additive.
- `kelly_mult ∈ {0.125, 0.25, 0.5}`, `min_edge ∈ {0, 0.01, 0.02, 0.03}`, chosen on validation P&L.
- Never use `volume_usd` as a feature. Hourly features use only points with `t ≤ price_ts`.
- The test set is scored once, after model and knobs are frozen.

```python
"""Feature table for model v2: data/horizons.csv + hourly paths + raw Gamma dump -> one DataFrame.

Every feature uses only information available at that row's horizon. volume_usd is lifetime volume
(includes trading after the horizon) and is deliberately not a feature.
"""
import json
import re
from pathlib import Path

import numpy as np
import pandas as pd

FEATURES = ["price_yes", "horizon_hours", "domain", "by_date", "n_legs", "book_share", "mom24", "mom72", "vol72"]
BY_DATE = re.compile(r"\b(by|before)\b\s+(the\s+end\s+of\s+)?"
                     r"(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec|\d|q[1-4]|end|202)", re.I)


def load_id_map(raw_dir):
    cache = raw_dir / "id_map.csv"
    if cache.exists():
        return pd.read_csv(cache, dtype={"gamma_id": str, "market_id": str})
    dump = sorted(raw_dir.glob("markets_*.json"))[-1]
    rows = [{"gamma_id": str(m["id"]), "market_id": m["conditionId"],
             "fees_on": str(m.get("feesEnabled")) == "True"} for m in json.loads(dump.read_text())]
    out = pd.DataFrame(rows)
    out.to_csv(cache, index=False)
    return out


def by_date(questions):
    return questions.fillna("").str.contains(BY_DATE).to_numpy()


def book_features(df):
    g = df.groupby(["event_slug", "horizon_hours"])["price_yes"]
    n = g.transform("size")
    return pd.DataFrame({"n_legs": n.astype(int),
                         "book_share": np.where(n > 1, df["price_yes"] / g.transform("sum"), 1.0)}, index=df.index)


def path_stats(t, p, at):
    keep = t <= at
    t, p = t[keep], p[keep]
    if len(t) == 0:
        return 0.0, 0.0, 0.0
    now = p[-1]

    def back(hours):
        i = np.searchsorted(t, at - hours * 3600, side="right") - 1
        return float(now - p[i]) if i >= 0 else 0.0

    w = p[t >= at - 72 * 3600]
    vol = float(np.std(np.diff(w))) if len(w) >= 12 else 0.0
    return back(24), back(72), vol


def build_table(data_dir, raw_dir):
    df = pd.read_csv(data_dir / "horizons.csv")
    df = df[df["outcome"].isin([0, 1]) & df["price_yes"].between(0, 1, inclusive="neither")].reset_index(drop=True)
    for c in ("resolved_at", "end_date", "price_ts"):
        df[c] = pd.to_datetime(df[c], utc=True, format="mixed")
    df["domain"] = df["domain"].fillna("other").str.lower()
    df["spread"] = pd.to_numeric(df["spread"], errors="coerce")
    df["by_date"] = by_date(df["question"])
    df = df.join(book_features(df))
    ids = load_id_map(raw_dir)
    df["fees_on"] = df["market_id"].map(ids.set_index("market_id")["fees_on"]).fillna(False).astype(bool)
    gamma = ids.set_index("market_id")["gamma_id"]
    stats = np.zeros((len(df), 3))
    for mid, rows in df.groupby("market_id").indices.items():
        f = raw_dir / "hist" / f"{gamma.get(mid)}.json"
        if not f.exists():
            continue
        pts = json.loads(f.read_text())
        t = np.array([x["t"] for x in pts], float)
        p = np.array([x["p"] for x in pts], float)
        order = np.argsort(t)
        t, p = t[order], p[order]
        for r in rows:
            stats[r] = path_stats(t, p, df.at[r, "price_ts"].timestamp())
    df[["mom24", "mom72", "vol72"]] = stats
    return df
```

```python
"""Model v2 playoff entrants. All share fit(df) -> self and predict(df) -> (q, lo, hi).

The band is the 10th-90th percentile of q over refits on events resampled with replacement.
Outside the tradable band the model is neither trained nor used: q = lo = hi = price.
"""
import numpy as np
import pandas as pd
from sklearn.ensemble import HistGradientBoostingClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler

TRADABLE = (0.03, 0.97)
DOMAINS = ["sports", "crypto", "politics", "economics", "culture", "science_tech", "geopolitics"]


def logit(p):
    p = np.clip(np.asarray(p, float), 0.01, 0.99)
    return np.log(p / (1 - p))


def sigmoid(x):
    return 1 / (1 + np.exp(-x))


def in_band(p):
    p = np.asarray(p, float)
    return (p >= TRADABLE[0]) & (p <= TRADABLE[1])


def anchor(q_raw, p, max_shift=0.7):
    return sigmoid(logit(p) + np.clip(logit(q_raw) - logit(p), -max_shift, max_shift))


def base_cols(df):
    lp = logit(df["price_yes"])
    lh = np.log(np.clip(df["horizon_hours"].to_numpy(float), 12, 720) / 24)
    return lp, lh


def full_matrix(df):
    lp, lh = base_cols(df)
    cols = [lp, lh, lp * lh, df["by_date"].to_numpy(float), lp * df["by_date"].to_numpy(float),
            np.log(df["n_legs"].to_numpy(float)), df["book_share"].to_numpy(float),
            df["mom24"].to_numpy(float), df["mom72"].to_numpy(float), df["vol72"].to_numpy(float),
            lp * df["vol72"].to_numpy(float)]
    cols += [(df["domain"] == d).to_numpy(float) for d in DOMAINS]
    return np.column_stack(cols)


class _Base:
    n_boot = 40

    def _x(self, df):
        raise NotImplementedError

    def _new(self):
        raise NotImplementedError

    def _q(self, est, df):
        return est.predict_proba(self._x(df))[:, 1]

    def fit(self, df, seed=0):
        df = df[in_band(df["price_yes"])].reset_index(drop=True)
        y = df["outcome"].to_numpy()
        self.main = self._new().fit(self._x(df), y)
        rng = np.random.default_rng(seed)
        groups = df.groupby("event_slug").indices
        ids = np.array(list(groups))
        self.boot = []
        for _ in range(self.n_boot):
            pick = np.concatenate([groups[i] for i in rng.choice(ids, len(ids))])
            self.boot.append(self._new().fit(self._x(df.iloc[pick]), y[pick]))
        return self

    def predict(self, df):
        p = df["price_yes"].to_numpy(float)
        q = self._q(self.main, df)
        qs = np.stack([self._q(m, df) for m in self.boot])
        lo, hi = np.percentile(qs, 10, axis=0), np.percentile(qs, 90, axis=0)
        lo, hi = np.minimum(lo, q), np.maximum(hi, q)
        b = in_band(p)
        return np.where(b, q, p), np.where(b, lo, p), np.where(b, hi, p)


class M0(_Base):
    """Current backend 'simple' calibration: logit(price), log horizon, interaction."""
    name = "M0"

    def _x(self, df):
        lp, lh = base_cols(df)
        return np.column_stack([lp, lh, lp * lh])

    def _new(self):
        return LogisticRegression(C=1.0, max_iter=2000)


class M1(_Base):
    """Logistic on all features, standardised."""
    name = "M1"

    def __init__(self, C=1.0):
        self.C = C

    def _x(self, df):
        return full_matrix(df)

    def _new(self):
        from sklearn.pipeline import make_pipeline
        return make_pipeline(StandardScaler(), LogisticRegression(C=self.C, max_iter=3000))


class M2(_Base):
    """Gradient boosting on all features, anchored to the market (log-odds shift <= 0.7)."""
    name = "M2"
    n_boot = 20

    def _x(self, df):
        return full_matrix(df)

    def _new(self):
        return HistGradientBoostingClassifier(max_depth=3, learning_rate=0.05, max_iter=300,
                                              min_samples_leaf=200, early_stopping=True,
                                              validation_fraction=0.15, random_state=0)

    def _q(self, est, df):
        return anchor(est.predict_proba(self._x(df))[:, 1], df["price_yes"].to_numpy(float))


def make(name, **kw):
    return {"M0": M0, "M1": M1, "M2": M2}[name](**kw)
```

```python
"""Trading rules for model v2: Polymarket taker fees, conservative edge, fractional Kelly,
one trade per market at the earliest qualifying horizon, held to resolution, and caps so the
whole book never stakes more than one starting bankroll at a time."""
import heapq

import numpy as np
import pandas as pd

from models import in_band

# docs.polymarket.com/trading/fees: fee per share = rate * p * (1 - p); makers pay nothing.
FEE_RATE = {"sports": 0.05, "crypto": 0.07, "politics": 0.04, "science_tech": 0.04, "economics": 0.05,
            "culture": 0.05, "other": 0.05, "geopolitics": 0.0}
PER_TRADE_CAP, PER_EVENT_CAP, BANKROLL_CAP = 0.05, 0.10, 1.0
EPS = 0.01


def trade_cost(p, spread, fees_on, domain, min_half=0.01):
    p = np.asarray(p, float)
    half = np.maximum(np.nan_to_num(np.asarray(spread, float), nan=0.0) / 2, min_half)
    rate = np.array([FEE_RATE.get(d, 0.05) for d in domain])
    return half + np.asarray(fees_on, bool) * rate * p * (1 - p)


def decide(p, q, lo, hi, cost, min_edge=0.0, kelly_mult=0.25):
    p = np.asarray(p, float)
    buy_yes = lo - (p + cost)
    buy_no = (p - cost) - hi
    band = in_band(p)
    action = np.where(band & (buy_yes > min_edge), "BUY_YES",
                      np.where(band & (buy_no > min_edge), "BUY_NO", "NO_TRADE"))
    qc = np.where(action == "BUY_YES", lo, np.where(action == "BUY_NO", hi, q))
    py, pn = np.clip(p + cost, EPS, 1 - EPS), np.clip(1 - p + cost, EPS, 1 - EPS)
    kelly = np.where(action == "BUY_YES", (qc - py) / (1 - py),
                     np.where(action == "BUY_NO", ((1 - qc) - pn) / (1 - pn), 0.0))
    kelly = np.clip(kelly * kelly_mult, 0, PER_TRADE_CAP)
    edge = np.where(action == "BUY_NO", buy_no, buy_yes)
    return action, kelly, edge


def payoff(action, p, cost, y):
    return np.where(action == "BUY_YES", y / np.clip(p + cost, EPS, 1) - 1,
                    np.where(action == "BUY_NO", (1 - y) / np.clip(1 - p + cost, EPS, 1) - 1, 0.0))


def earliest_trades(df, q, lo, hi, cost, min_edge, kelly_mult):
    t = df.assign(q=q, q_lo=lo, q_hi=hi, cost=cost)
    t["action"], t["kelly"], t["edge"] = decide(t["price_yes"].to_numpy(), q, lo, hi, cost, min_edge, kelly_mult)
    t = t[t["action"] != "NO_TRADE"].sort_values("horizon_hours", ascending=False)
    t = t.groupby("market_id").head(1).copy()
    t["ret"] = payoff(t["action"].to_numpy(), t["price_yes"].to_numpy(), t["cost"].to_numpy(), t["outcome"].to_numpy())
    t["entry_at"] = t["end_date"] - pd.to_timedelta(t["horizon_hours"], unit="h")
    return t


def apply_caps(trades):
    t = trades.sort_values(["entry_at", "market_id"]).copy()
    open_, used, by_event, stakes = [], 0.0, {}, []
    for r in t.itertuples():
        while open_ and open_[0][0] <= r.entry_at:
            _, stake, ev = heapq.heappop(open_)
            used -= stake
            by_event[ev] -= stake
        s = min(r.kelly, PER_EVENT_CAP - by_event.get(r.event_slug, 0.0), BANKROLL_CAP - used)
        s = max(s, 0.0)
        stakes.append(s)
        if s > 0:
            used += s
            by_event[r.event_slug] = by_event.get(r.event_slug, 0.0) + s
            heapq.heappush(open_, (r.resolved_at, s, r.event_slug))
    t["stake"] = stakes
    t["kelly"] = t["stake"]
    return t
```

```python
"""Model v2: playoff on validation, betting knobs tuned on validation, one out-of-time test.

  python integrations/model_v2/train.py [--data data] [--raw data/raw] [--out integrations/model_v2/results]
"""
import argparse
import json
import sys
from pathlib import Path

import numpy as np
import pandas as pd

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
sys.path.insert(0, str(HERE.parents[1] / "backend"))
import model as backend  # read-only: metric and report helpers  # noqa: E402

from decision import apply_caps, earliest_trades, trade_cost  # noqa: E402
from features import build_table  # noqa: E402
from models import in_band, make  # noqa: E402

CUTOFF = "2026-05-18"
KELLY_GRID = (0.125, 0.25, 0.5)
EDGE_GRID = (0.0, 0.01, 0.02, 0.03)
TIE = 0.0005


def split_time(df, cutoff):
    cut = pd.Timestamp(cutoff, tz="UTC")
    end = df.groupby("event_slug")["resolved_at"].transform("max")
    return df[end < cut], df[(end >= cut) & (df["resolved_at"] >= cut)]


def split_validation(train, frac=0.2):
    ends = train.groupby("event_slug")["resolved_at"].max().sort_values()
    val_events = set(ends.index[int(round(len(ends) * (1 - frac))):])
    v = train["event_slug"].isin(val_events)
    return train[~v], train[v]


def costs(df):
    return trade_cost(df["price_yes"], df["spread"], df["fees_on"], df["domain"])


CANDIDATES = [("M0", {}), ("M1", {"C": 0.1}), ("M1", {"C": 1.0}), ("M2", {})]


def playoff(fit, val):
    """Returns ((name, kwargs), scores). M1's C is picked here too; ties within TIE go to the simpler model."""
    vb = val[in_band(val["price_yes"])]
    y = vb["outcome"].to_numpy()
    scores = {"market": backend.log_loss(vb["price_yes"], y)}
    for name, kw in CANDIDATES:
        key = name + "".join(f"_C{v}" for v in kw.values())
        q, _, _ = make(name, **kw).fit(fit).predict(vb)
        scores[key] = backend.log_loss(q, y)
        print(f"  validation log loss {key}: {scores[key]:.4f} (market {scores['market']:.4f})", flush=True)
    best, best_score = CANDIDATES[0], scores["M0"]
    for name, kw in CANDIDATES[1:]:
        s = scores[name + "".join(f"_C{v}" for v in kw.values())]
        if s < best_score - TIE:
            best, best_score = (name, kw), s
    return best, {k: round(v, 5) for k, v in scores.items()}


def tune(model, val):
    q, lo, hi = model.predict(val)
    c = costs(val)
    grid = []
    for k in KELLY_GRID:
        for e in EDGE_GRID:
            t = apply_caps(earliest_trades(val, q, lo, hi, c, e, k))
            grid.append({"kelly_mult": k, "min_edge": e, "n_trades": int(len(t)),
                         "pnl": round(float((t["stake"] * t["ret"]).sum()), 4)})
    best = max(grid, key=lambda g: (g["pnl"], -g["kelly_mult"], g["min_edge"]))
    return best["kelly_mult"], best["min_edge"], grid
```

```python
    q, lo, hi = model.predict(live)

                      f"{p[i]:.0%}, net edge {edge[i] * 100:.1f}c after fees")
        out.append({"market_id": str(r.market_id), "question": r.question, "domain": r.domain,
                    "price_yes": round(float(p[i]), 4), "hours_to_resolution": float(r.hours_to_resolution),
                    "q_yes": round(float(q[i]), 4), "q_low": round(float(lo[i]), 4), "q_high": round(float(hi[i]), 4),
                    "edge": round(float(q[i] - p[i]), 4), "net_edge": round(float(edge[i]), 4), "cost": round(float(c[i]), 4),
                    "action": str(action[i]), "kelly_fraction": round(float(kelly[i]), 4), "reason": reason,
                    "url": r.url, "volume_usd": float(r.volume_usd)})
    trades = sorted((x for x in out if x["action"] != "NO_TRADE"), key=lambda x: -x["net_edge"])[:n_trades]
    no_trade = sorted((x for x in out if x["action"] == "NO_TRADE" and in_band(x["price_yes"])),
                      key=lambda x: -x["volume_usd"])[:n_no_trade]
    return trades + no_trade
```
